```javascript
// ── Anti-Ban: Human-Like Sender ─────────────────────────
const HumanLike = {
    // Delay log-normale (piu' realistico di uniform random)
    logNormalDelay(meanMs, stdMs) {
        // Box-Muller transform per distribuzione log-normale
        const u1 = Math.random();
        const u2 = Math.random();
        const z = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
        const delay = Math.exp(Math.log(meanMs) + z * (stdMs / meanMs));
        return Math.max(2000, Math.min(delay, meanMs * 3)); // clamp 2s - 3x media
    },

    // Typing indicator proporzionale alla lunghezza
    async simulateTyping(client, chatId, messageLength) {
        const chat = await client.getChatById(chatId);
        await chat.sendPresenceUpdate('composing');
        // ~50ms per carattere, min 2s, max 10s
        const typingMs = Math.max(2000, Math.min(10000, messageLength * 50 + Math.random() * 1500));
        await new Promise(r => setTimeout(r, typingMs));
    },

    // Recording indicator prima di voice note
    async simulateRecording(client, chatId, audioDurationSec) {
        const chat = await client.getChatById(chatId);
        await chat.sendPresenceUpdate('recording');
        // Simula registrazione: 80-120% della durata reale
        const recordMs = audioDurationSec * 1000 * (0.8 + Math.random() * 0.4);
        await new Promise(r => setTimeout(r, Math.min(recordMs, 30000))); // max 30s
    },

    // Check se il numero e' su WhatsApp PRIMA del primo contatto
    async checkOnWhatsApp(client, phone) {
        try {
            const chatId = phone.endsWith('@c.us') ? phone : `${phone}@c.us`;
            const isRegistered = await client.isRegisteredUser(chatId);
            return isRegistered;
        } catch (e) {
            log('WARN', `onWhatsApp check failed for ${phone}: ${e.message}`);
            return true; // assume true on error (don't block)
        }
    },

    // Clear typing/recording state
    async clearPresence(client, chatId) {
        try {
            const chat = await client.getChatById(chatId);
            await chat.clearState();
        } catch (_) {}
    },

    // Business hours enforcement (9-13, 15-19 IT)
    isAllowedToSend() {
        if (!TC.isBusinessHours()) {
            log('INFO', 'Anti-ban: fuori business hours, invio bloccato');
            return false;
        }
        return true;
    }
};
```
