```python
def compute_argos_grade(listing_id: str, db_path: str) -> dict:
    """
    Compute ARGOS GRADE A-E for a listing.

    Weights:
      35% CoVe confidence (from cove_results)
      20% Fraud flags (CLEAN=1.0, WARNING=0.5, SUSPICIOUS=0.0)
      15% Data completeness (how many of 7 fields are non-null in vehicle_listings)
      15% Photo count (0=0.0, 1-3=0.5, 4+=1.0 from vehicle_images)
      10% Recall status (0 recalls=1.0, 1-3=0.7, 4+=0.4 from NHTSA)
       5% KM history (always 0.5 for now — no free verification for DE)

    Grade mapping:
      A: score >= 0.85
      B: score >= 0.75
      C: score >= 0.65
      D: score >= 0.55
      E: score < 0.55

    Returns dict with: grade, score, components, recall_count, warranty_status
    """
```
